## PERF_AUDITOR — Lighthouse Runner

```python
# apps/agent-runtime/src/agents/qa/sub_agents/perf_auditor.py

import subprocess
import json

class PerfAuditorSubAgent(BaseSubAgent):
    async def run_lighthouse(self, url: str) -> dict:
        # Rodar Lighthouse via CLI (Node.js)
        cmd = [
            'node', '-e', f'''
const lighthouse = require('lighthouse');
const chromium = require('@sparticuz/chromium');
const puppeteer = require('puppeteer-core');

(async () => {{
    const browser = await puppeteer.launch({{
        args: chromium.args,
        executablePath: await chromium.executablePath(),
        headless: 'shell',
    }});
    const {{ lhr }} = await lighthouse('{url}', {{
        port: new URL(browser.wsEndpoint()).port,
        output: 'json',
        onlyCategories: ['performance', 'accessibility', 'best-practices', 'seo'],
    }});
    await browser.close();
    console.log(JSON.stringify({{
        performance: Math.round(lhr.categories.performance.score * 100),
        accessibility: Math.round(lhr.categories.accessibility.score * 100),
        bestPractices: Math.round(lhr.categories['best-practices'].score * 100),
        seo: Math.round(lhr.categories.seo.score * 100),
        issues: lhr.audits,
    }}));
}})();
'''
        ]

        result = subprocess.run(cmd, capture_output=True, text=True, timeout=120)
        return json.loads(result.stdout)
```

## Lógica de Decisão Final

```python
def evaluate_qa_results(sec: dict, perf: dict, content: dict) -> QADecision:
    issues = []

    # Segurança: qualquer CRITICAL ou HIGH = reprovar
    for issue in sec.get('issues', []):
        if issue['severity'] in ('CRITICAL', 'HIGH'):
            issues.append({'type': 'SECURITY', **issue})

    # Performance: abaixo do threshold = reprovar
    for metric, threshold in QA_THRESHOLDS.items():
        if metric.startswith('lighthouse_') and threshold is not None:
            key = metric.replace('lighthouse_', '')
            score = perf.get(key, 0)
            if score < threshold:
                issues.append({
                    'type': 'PERFORMANCE',
                    'severity': 'HIGH',
                    'message': f'{key}: {score} < {threshold}',
                })

    # Conteúdo: HIGH = reprovar, MEDIUM = avisar apenas
    for issue in content.get('issues', []):
        if issue['severity'] == 'HIGH':
            issues.append({'type': 'CONTENT', **issue})

    critical_issues = [i for i in issues if i.get('severity') == 'CRITICAL']
    high_issues = [i for i in issues if i.get('severity') == 'HIGH']

    if critical_issues or high_issues:
        return QADecision(
            approved=False,
            reason=f'{len(critical_issues)} críticos, {len(high_issues)} altos',
            issues=issues,
            action='send_to_builder_for_fix',
        )

    return QADecision(approved=True, issues=issues)
```

## Ciclo QA → Builder → QA

```
QA reprovar → Emitir QAFailed com lista de issues
→ Builder recebe e corrige (CODER sub-agente)
→ Novo build + novo deploy staging
→ QA roda novamente (max 3 ciclos)
→ Após 3 ciclos sem aprovação → HITL obrigatório com relatório detalhado
```
